## Phase 1: Component Coverage Foundation

### Mock Pattern Reference

The established pattern used across all Phase 1 tasks:

```tsx
// 1. Mocks before imports (vitest hoists them)
vi.mock('../../lib/supabase', () => ({ supabase: { from: () => mockChain } }));
vi.mock('../../hooks/useAuth', () => ({ useAuth: () => ({ user: mockUser, signOut: vi.fn() }) }));
vi.mock('react-hot-toast', () => ({ default: { error: vi.fn(), success: vi.fn() } }));
vi.mock('../../components/SEO', () => ({ SEO: ({ title }: { title: string }) => <title>{title}</title> }));
vi.mock('motion/react', () => ({ motion: { div: ({ children, ...p }: any) => <div {...p}>{children}</div> } }));

// 2. The chain-agnostic supabase mock (from BeatStore fix)
function createSupabaseMock(returnData: any[] = []) {
  const chain: any = {};
  chain.then = (cb: (val: any) => void) => cb({ data: returnData, error: null });
  return new Proxy(chain, {
    get(target, prop) {
      if (prop === 'then') return target.then;
      return () => chain;
    },
  });
}

// 3. Render with providers
function renderWithProviders(ui: React.ReactElement) {
  return render(
    <HelmetProvider>
      <MemoryRouter>{ui}</MemoryRouter>
    </HelmetProvider>
  );
}
```

### Task 2.1: Auth E2E workflows

**File:** `e2e/auth.spec.ts`

```ts
import { test, expect } from '@playwright/test';

test.describe('Artist Auth', () => {
  test('login with valid credentials → redirects to dashboard', async ({ page }) => {
    await page.goto('/artist/login');
    await page.fill('input[type="email"]', 'testartist@ncsound.test');
    await page.fill('input[type="password"]', 'test123');
    await page.click('button:has-text("Sign in to Portal")');
    await expect(page).toHaveURL(/\/artist\/dashboard/, { timeout: 10000 });
  });

  test('login with invalid credentials → shows error', async ({ page }) => {
    await page.goto('/artist/login');
    await page.fill('input[type="email"]', 'wrong@test.com');
    await page.fill('input[type="password"]', 'wrongpass');
    await page.click('button:has-text("Sign in to Portal")');
    await expect(page.locator('text=Invalid')).toBeVisible({ timeout: 10000 });
  });

  test('protected route redirects to login', async ({ page }) => {
    await page.goto('/artist/dashboard');
    await expect(page).toHaveURL(/\/artist\/login/);
  });
});

test.describe('Admin Auth', () => {
  test('login with valid admin credentials → redirects to dashboard', async ({ page }) => {
    await page.goto('/admin/login');
    await page.fill('input[type="email"]', 'testadmin@ncsound.test');
    await page.fill('input[type="password"]', 'test123');
    await page.click('button:has-text("Authenticate")');
    await expect(page).toHaveURL(/\/admin\/dashboard/, { timeout: 10000 });
  });
});

test.describe('Session Persistence', () => {
  test('session survives page reload', async ({ page }) => {
    await page.goto('/artist/login');
    await page.fill('input[type="email"]', 'testartist@ncsound.test');
    await page.fill('input[type="password"]', 'test123');
    await page.click('button:has-text("Sign in to Portal")');
    await page.waitForURL(/\/artist\/dashboard/);
    await page.reload();
    await expect(page).toHaveURL(/\/artist\/dashboard/);
  });
});
```
